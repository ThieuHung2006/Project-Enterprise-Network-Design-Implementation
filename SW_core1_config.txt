! =========================
! SW_core1 Configuration
! =========================

enable
configure terminal

hostname SW_core1

! -------------------------
! VLANs
! -------------------------
vlan 10
 name IT
vlan 11
 name HR
vlan 12
 name Accounting

! -------------------------
! Trunk Ports
! -------------------------
interface e0/3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10

interface e1/0
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 11,12

! -------------------------
! EtherChannel (LACP) to SW_core2
! -------------------------
interface range e0/0-2
 channel-group 1 mode active

interface port-channel 1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,11,12

! -------------------------
! L3 Interface to Router_DMZ
! -------------------------
interface e1/1
 no switchport
 no shutdown
 ip address 10.0.0.1 255.255.255.252

! -------------------------
! VLAN SVI + HSRP + DHCP relay
! -------------------------
interface vlan 10
 no shutdown
 ip address 192.168.10.2 255.255.255.0
 standby 10 priority 150
 standby 10 ip 192.168.10.1
 standby 10 preempt
 ip helper-address 192.168.99.10

interface vlan 11
 no shutdown
 ip address 192.168.11.2 255.255.255.0
 standby 11 priority 150
 standby 11 ip 192.168.11.1
 standby 11 preempt
 ip helper-address 192.168.99.10

interface vlan 12
 no shutdown
 ip address 192.168.12.2 255.255.255.0
 standby 12 priority 150
 standby 12 ip 192.168.12.1
 standby 12 preempt
 ip helper-address 192.168.99.10

! -------------------------
! Enable IP routing
! -------------------------
ip routing

! -------------------------
! OSPF Configuration
! -------------------------
router ospf 1
 router-id 1.1.1.1
 network 10.0.0.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.11.0 0.0.0.255 area 0
 network 192.168.12.0 0.0.0.255 area 0

end
write memory
